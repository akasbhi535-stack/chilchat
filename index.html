// ============================ // SEND MESSAGE // ============================ async function sendMessage(){ if(busy){ return; } const message = input.value.trim(); if(!message){ return; } busy = true; send.disabled = true; input.disabled = true; lastQuestion = message; addMessage( message, "user" ); input.value = ""; const thinking = createThinking(); try{ const response = await fetch( "/api/chat", { method:"POST", headers:{ "Content-Type":"application/json" }, body:JSON.stringify({ message:message, history:history.slice(-20) }) } ); const data = await response.json(); if(!response.ok){ throw new Error( data.error || "Server error " + response.status ); } thinking.remove(); const reply = data.reply; if(!reply){ throw new Error( "AI response nahi mila." ); } addMessage( reply, "bot" ); // Save user message history.push({ role:"user", text:message }); // Save AI response history.push({ role:"bot", text:reply }); saveHistory(); scrollToBottom(); }catch(error){ thinking.remove(); addMessage( "Sorry 😕\n\n" + error.message, "bot" ); } busy = false; send.disabled = false; input.disabled = false; input.focus(); }
